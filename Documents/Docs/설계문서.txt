================================================================================
           신규 게임 설계 문서 (GDD + TDD 통합)
           RE001 디자인 + SlimeMaster 코딩 스타일 기반
================================================================================

이 문서는 RE001 레퍼런스 게임의 디자인을 목표로 삼고,
SlimeMaster에서 검증된 코딩 패턴을 그대로 적용하여 개발하기 위한 통합 설계서입니다.

PART 1. 게임 개요
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  장르     : 2D 횡스크롤 스토리 액션 RPG (모바일)
  해상도   : 1440*3040
  엔진     : Unity (URP)
  저장방식 : JSON 직렬화 (Newtonsoft.Json) - SlimeMaster 동일

  핵심 루프
  ┌──────────────────────────────────────────────────────────┐
  │  마을 허브                                               │
  │   ├─ NPC 대화 → 퀘스트 수주 / 기능 해금                │
  │   ├─ 대장간  → 장비 강화 (FORGE)                       │
  │   └─ 연금공방 → 스킬/물약 제작 의뢰                    │
  │         ↓                                               │
  │  탐험 구역 입장 (숲, 사막 등 구역별)                    │
  │   ├─ 횡스크롤 자동 전투                                 │
  │   ├─ 몬스터 처치 → 경험치 / 골드 / 아이템 드롭         │
  │   └─ 야생 몬스터 조우 → 미니몬 포획 시도               │
  │         ↓                                               │
  │  귀환 → 강화 → 다음 구역 진입                          │
  └──────────────────────────────────────────────────────────┘

PART 4. 세이브 데이터 구조 (SlimeMaster GameData 확장)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  [Serializable]
  public class GameData
  {
      // 기본 (SlimeMaster 동일)
      public string  PlayerName;
      public int     Level;
      public float   Exp;
      public int     Gold;
      public int     Crystal;           // 크리스탈 (RE001의 두 번째 재화)

      // 장비 (SlimeMaster 동일)
      public List<Equipment> OwnedEquipments = new();
      public Dictionary<EquipSlotType, Equipment> EquippedEquipments = new();

      // 인벤토리 [신규 - RE001 소지품]
      public List<ItemData> Inventory = new();

      // 미니몬 [신규]
      public List<MinMonSaveData> OwnedMinMons = new();
      public int CompanionMinMonId = -1;     // 현재 동행 중인 미니몬

      // 퀘스트/해금 상태 [신규]
      public List<int> CompletedQuestIds = new();
      public List<string> UnlockedTabs   = new();   // 해금된 하단 탭 목록
      public List<int> UnlockedAreaIds   = new();   // 입장 가능한 탐험 구역

      // 마물 도감 [신규]
      public List<int> DiscoveredMonsterIds = new();
  }

  // 미니몬 세이브 데이터 [신규]
  [Serializable]
  public class MinMonSaveData
  {
      public int  DataId;
      public int  TameCount;        // 포획 시도 횟수 (동기화% 계산용)
      public bool IsCompanion;
  }

  // 장비 런타임 데이터 (SlimeMaster Equipment 확장)
  [Serializable]
  public class Equipment
  {
      public int    DataId;
      public int    EnhanceLevel;   // +1, +2... 강화 단계 [신규]
      public int    Durability;     // 내구도 [신규]
      public bool   IsEquipped;
  }

  파일명                    딕셔너리 키   내용
  ─────────────────────────────────────────────────────────────────────
  CreatureData.json         int           플레이어/몬스터 기본 스탯
  EquipmentData.json        int           장비 종류, 기본 능력치, 슬롯 타입
  EquipEnhanceData.json     (int,int)     장비ID + 강화단계별 능력치  [신규]
  SkillData.json            int           스킬 쿨타임, 데미지 배율
  ItemData.json             int           소지품 종류, 등급, 설명
  AreaData.json             int           탐험 구역 정보, 등장 몬스터 [신규]
  QuestData.json            int           퀘스트 조건, 보상, 선행 퀘스트 [신규]
  DialogueData.json         int           대화 ID, 화자, 텍스트 배열  [신규]
  MinMonData.json           int           미니몬 기본 스탯, 등급, 스킬 [신규]
  ForgeRecipeData.json      int           강화 재료 레시피             [신규]
  CraftRecipeData.json      int           연금공방 제작 레시피         [신규]
  LevelData.json            int           레벨별 필요 경험치
  ─────────────────────────────────────────────────────────────────────

  // 예시: EquipEnhanceData 로더
  public class EquipEnhanceDataLoader : ILoader<(int, int), EquipEnhanceData>
  {
      public List<EquipEnhanceData> EnhanceList = new();
      public Dictionary<(int, int), EquipEnhanceData> MakeDict()
          => EnhanceList.ToDictionary(x => (x.EquipDataId, x.Level));
  }

  // GameManager 장비 강화 메서드
  public bool ForgeEquipment(Equipment equip, int materialId)
  {
      var recipe = Managers.Data.ForgeRecipeDic[(equip.DataId, equip.EnhanceLevel + 1)];
      if (!HasMaterial(materialId, recipe.RequiredCount)) return false;

      RemoveMaterial(materialId, recipe.RequiredCount);
      equip.EnhanceLevel++;
      equip.Durability = Managers.Data.EquipDic[equip.DataId].MaxDurability;
      SaveGame();
      OnEquipmentForged?.Invoke(equip);   // 이벤트로 UI에 통보
      return true;
  }

  // 동기화(%) = TameCount / MaxTameCount * 100
  // 스탯 = BaseAtk * (1 + SyncRate * 성장배율)
  public float GetSyncRate(int dataId)
  {
      var save = GetSaveData(dataId);
      var data = Managers.Data.MinMonDic[dataId];
      return (float)save.TameCount / data.MaxTameCount;
  }

  // 동행 설정 (1마리만 가능)
